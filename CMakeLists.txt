cmake_minimum_required(VERSION 3.16)
project(ready_queue LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(scheduler INTERFACE)
target_include_directories(scheduler INTERFACE include)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(scheduler INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(ready-queue src/main.cpp)
target_link_libraries(ready-queue PRIVATE scheduler)
enable_testing()
add_executable(scheduler-tests tests/scheduler_test.cpp)
target_link_libraries(scheduler-tests PRIVATE scheduler)
add_test(NAME scheduler-tests COMMAND scheduler-tests)
add_test(NAME example COMMAND ready-queue ${CMAKE_CURRENT_SOURCE_DIR}/examples/build.tasks)
set_tests_properties(example PROPERTIES PASS_REGULAR_EXPRESSION "lint\ncompile\ntest\npackage")
